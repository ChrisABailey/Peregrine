# peregrine_core — the Peregrine core as one shared library (extensions plan
# EX-D8): every static library in DeskKit's link closure, linked whole, so a
# `native` plugin and the app share one copy of each process-wide table.
# Consumers get the closure's headers and FV_SDK_BUILD_ID, not its archives.
# Also writes the build-tree CMake package (Peregrine::core) that out-of-tree
# plugins find with find_package(Peregrine CONFIG).
include(${CMAKE_SOURCE_DIR}/port/cmake/FvwStaticClosure.cmake)

add_library(peregrine_core_closure INTERFACE)
target_link_libraries(peregrine_core_closure INTERFACE fv_deskkit)
fvw_collect_static_libs(peregrine_core_closure PEREGRINE_CORE_STATIC_LIBS)
# Its tables duplicate ImageLib's; the normal link below takes one of them.
list(REMOVE_ITEM PEREGRINE_CORE_STATIC_LIBS fv_geotiff_names)

add_library(peregrine_core SHARED core/peregrine_core.cpp)
add_library(Peregrine::core ALIAS peregrine_core)
set(_whole "")
foreach(_l IN LISTS PEREGRINE_CORE_STATIC_LIBS)
  if(APPLE)
    list(APPEND _whole "LINKER:-force_load,$<TARGET_FILE:${_l}>")
  else()
    list(APPEND _whole "$<TARGET_FILE:${_l}>")
  endif()
endforeach()
if(APPLE)
  target_link_options(peregrine_core PRIVATE ${_whole})
else()
  target_link_options(peregrine_core PRIVATE "LINKER:--whole-archive" ${_whole} "LINKER:--no-whole-archive")
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  # Exported, the third-party copies inside the core (libpng 1.2, libjpeg 6b,
  # zlib, expat, libtiff) would interpose the system libraries GTK loads.
  # Plugins reach none of them directly, so they stay local to the core.
  set(_third_party_names fv_png fv_jpeg expat zlibstatic zlib tiff)
  set(_third_party "")
  foreach(_l IN LISTS PEREGRINE_CORE_STATIC_LIBS)
    # The closure names some libraries by alias (TIFF::tiff).
    get_target_property(_real ${_l} ALIASED_TARGET)
    if(NOT _real)
      set(_real ${_l})
    endif()
    if(_real IN_LIST _third_party_names)
      list(APPEND _third_party "$<TARGET_FILE_NAME:${_real}>")
    endif()
  endforeach()
  list(REMOVE_DUPLICATES _third_party)
  if(_third_party)
    string(JOIN ":" _excluded ${_third_party})
    target_link_options(peregrine_core PRIVATE "LINKER:--exclude-libs,${_excluded}")
  endif()
endif()
add_dependencies(peregrine_core ${PEREGRINE_CORE_STATIC_LIBS})
# The normal link brings the closure's system libraries and frameworks; its
# archives add nothing after the whole-archive pass above.
target_link_libraries(peregrine_core PRIVATE fv_deskkit
  INTERFACE $<COMPILE_ONLY:fv_deskkit>)
target_include_directories(peregrine_core PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<BUILD_INTERFACE:${FVW_SDK_GEN_DIR}>)
target_compile_definitions(peregrine_core INTERFACE FV_SDK_BUILD_ID="${FVW_SDK_BUILD_ID}")
set_target_properties(peregrine_core PROPERTIES
  OUTPUT_NAME peregrine_core
  VERSION ${PEREGRINE_VERSION}
  SOVERSION 0
  INSTALL_NAME_DIR "@rpath")

# The build-tree package: find_package(Peregrine CONFIG) with Peregrine_DIR
# set to this directory gives an out-of-tree plugin Peregrine::core.
file(GENERATE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/PeregrineConfig.cmake CONTENT
"# Generated by port/sdk/CMakeLists.txt. The Peregrine core of the build tree
# ${CMAKE_BINARY_DIR}, build id ${FVW_SDK_BUILD_ID}.
if(NOT TARGET Peregrine::core)
  add_library(Peregrine::core SHARED IMPORTED)
  set_target_properties(Peregrine::core PROPERTIES
    IMPORTED_LOCATION \"$<TARGET_FILE:peregrine_core>\"
    IMPORTED_SONAME \"$<TARGET_SONAME_FILE_NAME:peregrine_core>\"
    INTERFACE_INCLUDE_DIRECTORIES \"$<TARGET_PROPERTY:peregrine_core,INTERFACE_INCLUDE_DIRECTORIES>\"
    INTERFACE_COMPILE_DEFINITIONS \"FV_SDK_BUILD_ID=\\\"${FVW_SDK_BUILD_ID}\\\"\"
    INTERFACE_COMPILE_FEATURES cxx_std_17)
endif()
set(Peregrine_SDK_BUILD_ID \"${FVW_SDK_BUILD_ID}\")
")
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/PeregrineConfigVersion.cmake
"set(PACKAGE_VERSION \"${PEREGRINE_VERSION}\")
set(PACKAGE_VERSION_COMPATIBLE TRUE)
if(PACKAGE_FIND_VERSION STREQUAL PACKAGE_VERSION)
  set(PACKAGE_VERSION_EXACT TRUE)
endif()
")

# The Python bridge; Python stays optional, so it is built only where pybind11
# found an interpreter to embed.
if(TARGET pybind11::embed)
  add_subdirectory(python)
endif()

  target_link_libraries(peregrine_native_plugin_test PRIVATE peregrine_core GTest::gtest_main)
  target_include_directories(peregrine_native_plugin_test PRIVATE ${CMAKE_SOURCE_DIR}/port/DeskKit/test)
  add_library(fv_test_stale_native SHARED test/plugins/stale_native.cpp)
  target_link_libraries(fv_test_stale_native PRIVATE peregrine_core)
  set_target_properties(fv_test_stale_native PROPERTIES CXX_VISIBILITY_PRESET hidden)
  add_dependencies(peregrine_native_plugin_test fv_test_stale_native nativerings_pgpkg)
  target_compile_definitions(peregrine_native_plugin_test PRIVATE
    FV_NATIVERINGS_PGPKG="${CMAKE_BINARY_DIR}/port/sdk/examples/rangerings-cpp/com.example.nativerings.pgpkg"
    FV_NATIVERINGS_LIB="$<TARGET_FILE:nativerings>"
    FV_STALE_NATIVE="$<TARGET_FILE:fv_test_stale_native>")
  gtest_discover_tests(peregrine_native_plugin_test)

  # The python kind, against the shared core like the app. The bridge-backed

  target_link_libraries(peregrine_python_plugin_test PRIVATE peregrine_core GTest::gtest_main)
  target_include_directories(peregrine_python_plugin_test PRIVATE ${CMAKE_SOURCE_DIR}/port/DeskKit/test)
  add_dependencies(peregrine_python_plugin_test pyrings_pgpkg rangerings_pgpkg)
  target_compile_definitions(peregrine_python_plugin_test PRIVATE
    FV_PYRINGS_PGPKG="${CMAKE_BINARY_DIR}/port/sdk/examples/rangerings-py/com.example.pyrings.pgpkg"
    FV_RANGERINGS_PGPKG="${CMAKE_BINARY_DIR}/port/sdk/examples/rangerings-c/com.example.rangerings.pgpkg")
  if(TARGET peregrine_python)
    add_dependencies(peregrine_python_plugin_test peregrine_python)
    target_compile_definitions(peregrine_python_plugin_test PRIVATE
      FV_PYTHON_BRIDGE="$<TARGET_FILE:peregrine_python>")
  endif()
  gtest_discover_tests(peregrine_python_plugin_test)
endif()
